/**
 * Chino PC Master — Apps Script Bingo Digital (multi-proyecto, versión Beta 2).
 *
 * Un solo Web App:
 *   - Spreadsheet de registro (SPREADSHEET_ID): hoja «Bingos», una fila por bingo.
 *   - Por cada bingo: un Spreadsheet propio y una carpeta en Drive (comprobantes y arte del cartón),
 *     creados al momento de crear el bingo.

 *   MASTER_ADMIN_PIN    — llave maestra del hub #bingo (obligatoria)
 *   ROOT_FOLDER_ID      — carpeta de Drive donde se crean las carpetas de cada bingo (opcional)
 *   VIVO_SPREADSHEET_ID — la crea «instalarBingo» (no la escribas a mano)
 *
 * Instalación: pega este archivo en Código.js, define las propiedades, ejecuta UNA VEZ «instalarBingo»
 * desde el editor (crea el Spreadsheet público en vivo) y despliega como Aplicación web
 * (Ejecutar como: Yo · Acceso: Cualquiera). La URL /exec va en index.html como window.CPM_BINGO_GAS_URL.
 */

function doGet() {
  return jsonOut_({ status: "SUCCESS", data: { service: "bingo_multi", version: API_VERSION_ } });
}

function doPost(e) {
  try {
    var data = parseBody_(e);
    var action = String(data.action || "").trim();
    if (!action) return jsonOut_({ status: "ERROR", message: "Acción vacía." });
    if (action === "ping") return ok_({ pong: true, version: API_VERSION_ });

    if (action.indexOf("super_") === 0) {
      assertMaster_(data.masterPin);
      if (action === "super_list") return ok_(superList_());
      if (action === "super_create") return ok_(conLock_(function () { return superCreate_(data); }));
      if (action === "super_update") return ok_(conLock_(function () { return superUpdate_(data); }));
      if (action === "super_delete") return ok_(conLock_(function () { return superDelete_(data); }));
      return fail_("Acción del hub no reconocida («" + action + "»).");
    }

    /* Público: compra, jugador, cartón suelto y estado en vivo */
    if (action === "get_compra") return ok_(getCompra_(bingoPorPublico_(data.pub)));
    if (action === "submit_compra") {
      var bc = bingoPorPublico_(data.pub);
      return ok_(conLock_(function () { return submitCompra_(bc, data); }));
    }
    if (action === "get_player") return ok_(getPlayer_(data.token));
    if (action === "get_card") return ok_(getCard_(bingoPorPublico_(data.pub)));
    if (action === "estado") return ok_(estadoVivo_(bingoPorPublico_(data.pub)));
    if (action === "get_arte") return ok_(getArte_(bingoPorPublico_(data.pub)));

    /* Vendedor o encargado de sector */
    if (action.indexOf("sector_") === 0) {
      var ctxS = bingoPorSector_(data.hash);
      if (action === "sector_resolve") return ok_(sectorResolve_(ctxS));
      if (action === "sector_vender") return ok_(conLock_(function () { return venderPresencial_(ctxS.bingo, data, ctxS.sector, ctxS.encargado); }));
      if (action === "sector_comprobante") return ok_(getComprobante_(ctxS.bingo, data.entrada_id, ctxS.sector));
      if (action === "sector_mark_sent") return ok_(conLock_(function () { return markSent_(ctxS.bingo, data, ctxS.sector); }));
      return fail_("Acción de sector no reconocida.");
    }

    /* Juego: organizador o cantador */
    if (action.indexOf("game_") === 0) {
      var bg = bingoPorAdminOCantador_(data.hash);
      if (action === "game_get") return ok_(gameGet_(bg));
      if (action === "game_save_ronda") return ok_(conLock_(function () { return gameSaveRonda_(bg, data); }));
      if (action === "game_delete_ronda") return ok_(conLock_(function () { return gameDeleteRonda_(bg, data); }));
      if (action === "game_abrir") return ok_(conLock_(function () { return gameAbrir_(bg, data); }));
      if (action === "game_cantar") return ok_(conLock_(function () { return gameCantar_(bg, data); }));
      if (action === "game_deshacer") return ok_(conLock_(function () { return gameDeshacer_(bg, data); }));
      if (action === "game_mover_ronda") return ok_(conLock_(function () { return gameMoverRonda_(bg, data); }));
      if (action === "game_quitar_ganador") return ok_(conLock_(function () { return gameQuitarGanador_(bg, data); }));
      if (action === "game_ganador") return ok_(conLock_(function () { return gameGanador_(bg, data); }));
      if (action === "game_cerrar") return ok_(conLock_(function () { return gameCerrar_(bg, data); }));
      if (action === "game_buscar") return ok_(gameBuscar_(bg, data));
      return fail_("Acción de juego no reconocida.");
    }

    /* Organizador */
    var b = bingoPorAdmin_(data.hash);
    if (action === "resolve_admin") return ok_(resolveAdmin_(b));
    if (action === "update_config") return ok_(conLock_(function () { return updateConfig_(b, data.config || {}); }));
    if (action === "upload_arte") return ok_(conLock_(function () { return uploadArte_(b, data); }));
    if (action === "delete_arte") return ok_(conLock_(function () { return updateConfig_(b, { arte_file_id: "" }); }));
    if (action === "upload_centro") return ok_(conLock_(function () { return uploadCentro_(b, data); }));
    if (action === "delete_centro") return ok_(conLock_(function () { return updateConfig_(b, { centro_file_id: "" }); }));
    if (action === "save_precios") return ok_(conLock_(function () { return savePrecios_(b, data.precios || []); }));
    if (action === "save_sectores") return ok_(conLock_(function () { return saveSectores_(b, data.sectores || []); }));
    if (action === "list_entradas") return ok_(listEntradas_(b));
    if (action === "get_comprobante") return ok_(getComprobante_(b, data.entrada_id));
    if (action === "approve_entrada") return ok_(conLock_(function () { return approveEntrada_(b, data); }));
    if (action === "reject_entrada") return ok_(conLock_(function () { return setEstadoEntrada_(b, data.entrada_id, "Rechazada", data.nota, true); }));
    if (action === "return_entrada") return ok_(conLock_(function () { return setEstadoEntrada_(b, data.entrada_id, "Pendiente", data.nota, true); }));
    if (action === "complete_entrada") return ok_(conLock_(function () { return setEstadoEntrada_(b, data.entrada_id, "Completada", "", false); }));
    if (action === "reopen_entrada") return ok_(conLock_(function () { return setEstadoEntrada_(b, data.entrada_id, "Aprobada", "", false); }));
    if (action === "mark_sent") return ok_(conLock_(function () { return markSent_(b, data, null); }));
    if (action === "send_email") return ok_(conLock_(function () { return sendEmail_(b, data.entrada_id); }));
    if (action === "vender_presencial") return ok_(conLock_(function () { return venderPresencial_(b, data, null); }));
    if (action === "ampliar_lote") return ok_(conLock_(function () { return ampliarLote_(b, data.cantidad); }));
    if (action === "list_cartones") return ok_(listCartones_(b));
    return fail_("Acción no reconocida («" + action + "»).");
  } catch (err) {
    return jsonOut_({ status: "ERROR", message: String(err && err.message ? err.message : err) });
  }
}

function ok_(data) { return jsonOut_({ status: "SUCCESS", data: data }); }
function fail_(msg) { return jsonOut_({ status: "ERROR", message: msg }); }

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function parseBody_(e) {
  if (!e) return {};
  var raw = e.postData && e.postData.contents ? String(e.postData.contents) : "";
  if (raw && raw.charAt(0) !== "{" && e.parameter && e.parameter.payload) raw = String(e.parameter.payload);
  if (!raw && e.parameter && e.parameter.payload) raw = String(e.parameter.payload);
  if (!raw) return e.parameter || {};
  try {
    return JSON.parse(raw);
  } catch (x) {
    var m = raw.match(/(?:^|&)payload=([^&]*)/);
    if (m) return JSON.parse(decodeURIComponent(m[1].replace(/\+/g, " ")));
    throw new Error("Cuerpo de la petición no es JSON.");
  }
}

function conLock_(fn) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) throw new Error("El servidor está ocupado. Intenta de nuevo en unos segundos.");
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

function hoyIso_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"); }
function ahora_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); }

function fechaIso_(v) {
  if (!v) return "";
  if (Object.prototype.toString.call(v) === "[object Date]") {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  return String(v).trim();
}

/** Sheets convierte «19:00» en una fecha de 1899: se devuelve como hora. */
function valorTexto_(v) {
  if (Object.prototype.toString.call(v) !== "[object Date]") return String(v);
  if (v.getFullYear() < 1901) return Utilities.formatDate(v, Session.getScriptTimeZone(), "HH:mm");
  return fechaIso_(v);
}

var MESES_ = ["ene", "feb", "mar", "abr", "may", "jun", "jul", "ago", "sep", "oct", "nov", "dic"];

/** «2026-11-15» + «19:00» → «15 de nov de 2026 a las 19:00». */
function fechaLarga_(fecha, hora) {
  var m = String(fecha || "").match(/^(\d{4})-(\d{2})-(\d{2})/);
  var h = String(hora || "").match(/^(\d{1,2}):(\d{2})/);
  var t = m ? Number(m[3]) + " de " + MESES_[Number(m[2]) - 1] + " de " + m[1] : String(fecha || "");
  if (h) t += (t ? " a las " : "") + ("0" + h[1]).slice(-2) + ":" + h[2];
  return t;
}

/** Los emojis se ven como cuadros en muchos lectores de correo: se quitan del asunto y del cuerpo. */
function sinEmojis_(t) {
  return String(t || "").replace(/[\u{1F000}-\u{1FAFF}\u{2600}-\u{27BF}\u{2B00}-\u{2BFF}\u{1F1E6}-\u{1F1FF}\u{FE0F}\u{200D}\u{20E3}]/gu, "");
}

function correoClave_(c) { return String(c || "").trim().toLowerCase(); }
function correoValido_(c) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(String(c || "").trim()); }

function bool_(v) {
  if (v === true) return true;
  var t = String(v || "").trim().toLowerCase();
  return t === "true" || t === "1" || t === "si" || t === "sí";
}

function randomHash_(len) {
  var chars = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789";
  var out = "";
  for (var i = 0; i < len; i++) out += chars.charAt(Math.floor(Math.random() * chars.length));
  return out;
}

function ensureSheet_(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold");
    sh.setFrozenRows(1);
  }
  return sh;
}

function readObjects_(sh) {
  var values = sh.getDataRange().getValues();
  if (values.length < 2) return [];
  var headers = values[0].map(function (h) { return String(h).trim(); });
  var out = [];
  for (var r = 1; r < values.length; r++) {
    var o = { _row: r + 1 };
    for (var c = 0; c < headers.length; c++) o[headers[c]] = values[r][c];
    out.push(o);
  }
  return out;
}

function writeObject_(sh, headers, obj, rowNumber) {
  var row = headers.map(function (h) { return obj[h] === undefined || obj[h] === null ? "" : obj[h]; });
  if (rowNumber) sh.getRange(rowNumber, 1, 1, headers.length).setValues([row]);
  else sh.appendRow(row);
}

function nextId_(items, key) {
  var max = 0;
  items.forEach(function (it) { max = Math.max(max, Number(it[key]) || 0); });
  return max + 1;
}

/** Acepta "5, 8, 10-14" → [5, 8, 10, 11, 12, 13, 14] */
function parseSeries_(input) {
  var out = [];
  var seen = {};
  var parts = Array.isArray(input) ? input : String(input || "").split(/[,\s;]+/);
  parts.forEach(function (p) {
    var t = String(p).trim();
    if (!t) return;
    var m = t.match(/^(\d+)\s*-\s*(\d+)$/);
    var a = m ? Number(m[1]) : Number(t);
    var z = m ? Number(m[2]) : Number(t);
    if (!isFinite(a) || !isFinite(z)) throw new Error("Serie no válida: «" + t + "».");
    if (z < a) { var tmp = a; a = z; z = tmp; }
    if (z - a > 2000) throw new Error("Rango demasiado amplio: «" + t + "» (máximo 2000 por rango).");
    for (var n = a; n <= z; n++) {
      if (!seen[n]) { seen[n] = 1; out.push(n); }
    }
  });
  return out;
}

function render_(tpl, vars) {
  return String(tpl || "").replace(/\{\{(\w+)\}\}/g, function (_, k) {
    return vars[k] === undefined ? "" : String(vars[k]);
  });
}

/** Ejecútala UNA VEZ desde el editor: crea el Spreadsheet público «Bingo en vivo». */
function instalarBingo() {
  var props = PropertiesService.getScriptProperties();
  var reg = registroSs_();
  ensureSheet_(reg, "Bingos", BINGOS_HEADERS_);
  var vivoId = prop_("VIVO_SPREADSHEET_ID");
  if (!vivoId) {
    var vivo = SpreadsheetApp.create("Bingo en vivo (público, solo bolas cantadas)");
    var sh = vivo.getSheets()[0];
    sh.setName("vivo");
    sh.getRange(1, 1, 1, 3).setValues([["public_id", "estado_json", "actualizado"]]);
    DriveApp.getFileById(vivo.getId()).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    var root = prop_("ROOT_FOLDER_ID");
    if (root) DriveApp.getFileById(vivo.getId()).moveTo(DriveApp.getFolderById(root));
    props.setProperty("VIVO_SPREADSHEET_ID", vivo.getId());
    vivoId = vivo.getId();
  }
  Logger.log("Listo. VIVO_SPREADSHEET_ID = " + vivoId);
}

/* ======================= Registro de bingos ======================= */

function bingosSheet_() { return ensureSheet_(registroSs_(), "Bingos", BINGOS_HEADERS_); }

function listBingos_() {
  return readObjects_(bingosSheet_()).map(function (b) {
    b.activo = bool_(b.activo);
    b.fecha_archivo = fechaIso_(b.fecha_archivo);
    b.vencido = Boolean(b.fecha_archivo) && hoyIso_() > b.fecha_archivo;
    return b;
  });
}

function vigente_(b) { return b && b.activo && !b.vencido; }

function bingoPor_(campo, valor, msg) {
  var v = String(valor || "").trim();
  if (!v) throw new Error(msg);
  var found = listBingos_().filter(function (b) { return String(b[campo]) === v; })[0];
  if (!found) throw new Error(msg);
  if (!vigente_(found)) throw new Error("Este bingo ya no está activo.");
  return found;
}

function bingoPorAdmin_(hash) { return bingoPor_("hash_admin", hash, "Link de organizador no válido."); }
function bingoPorPublico_(pub) { return bingoPor_("public_id", pub, "Bingo no encontrado."); }

function bingoPorAdminOCantador_(hash) {
  var v = String(hash || "").trim();
  var found = listBingos_().filter(function (b) { return v && (b.hash_admin === v || b.hash_cantador === v); })[0];
  if (!found) throw new Error("Link no válido.");
  if (!vigente_(found)) throw new Error("Este bingo ya no está activo.");
  found._rol = found.hash_admin === v ? "organizador" : "cantador";
  return found;
}

/** El link de sector lleva el public_id del bingo al inicio («pub.xxxx»): se abre solo ese Spreadsheet. */
function bingoPorSector_(hash) {
  var v = String(hash || "").trim();
  var b = bingoPorPublico_(v.split(".")[0]);
  var sectores = sectores_(b);
  for (var j = 0; j < sectores.length; j++) {
    var s = sectores[j];
    if (!s.activo) continue;
    if (s.hash === v) return { bingo: b, sector: s, encargado: null };
    for (var k = 0; k < s.encargados.length; k++) {
      if (s.encargados[k].hash === v && s.encargados[k].activo) return { bingo: b, sector: s, encargado: s.encargados[k] };
    }
  }
  throw new Error("Link de sector no válido o desactivado.");
}

function bingoPublico_(b) {
  return {
    bingo_id: b.bingo_id, nombre: b.nombre, public_id: b.public_id, hash_admin: b.hash_admin,
    hash_cantador: b.hash_cantador, created_at: fechaIso_(b.created_at), activo: b.activo,
    fecha_archivo: b.fecha_archivo, vencido: b.vencido, compartido_con: b.compartido_con,
    spreadsheet_url: b.spreadsheet_id ? "https://docs.google.com/spreadsheets/d/" + b.spreadsheet_id + "/edit" : "",
    folder_url: b.folder_id ? "https://drive.google.com/drive/folders/" + b.folder_id : ""
  };
}

function superList_() {
  return { bingos: listBingos_().map(bingoPublico_), version: API_VERSION_ };
}

function superCreate_(data) {
  var nombre = String(data.nombre || "").trim();
  if (!nombre) throw new Error("Escribe el nombre del bingo.");
  var sh = bingosSheet_();
  var bingos = readObjects_(sh);
  var id = nextId_(bingos, "bingo_id");
  var usados = {};
  bingos.forEach(function (x) { usados[x.public_id] = 1; usados[x.hash_admin] = 1; usados[x.hash_cantador] = 1; });
  function unico(len) { var h; do { h = randomHash_(len); } while (usados[h]); usados[h] = 1; return h; }

  var ss = SpreadsheetApp.create("Bingo " + id + " — " + nombre);
  Object.keys(SHEETS_).forEach(function (name) { ensureSheet_(ss, name, SHEETS_[name]); });
  var hoja1 = ss.getSheetByName("Hoja 1") || ss.getSheetByName("Sheet1");
  if (hoja1 && ss.getSheets().length > 1) ss.deleteSheet(hoja1);

  var rootId = prop_("ROOT_FOLDER_ID");
  var root = rootId ? DriveApp.getFolderById(rootId) : DriveApp.getRootFolder();
  var folder = root.createFolder("Bingo " + id + " — " + nombre);
  folder.createFolder("comprobantes");
  folder.createFolder("diseño");
  DriveApp.getFileById(ss.getId()).moveTo(folder);

  var cfg = {};
  Object.keys(CONFIG_DEFAULTS_).forEach(function (k) { cfg[k] = CONFIG_DEFAULTS_[k]; });
  cfg.nombre = nombre;
  cfg.fecha = String(data.fecha || "").trim();
  cfg.sitio_url = String(data.sitio_url || "").trim();
  var cfgSh = ss.getSheetByName("config");
  cfgSh.getRange(2, 1, Object.keys(cfg).length, 2).setValues(Object.keys(cfg).map(function (k) { return [k, cfg[k]]; }));

  var preciosSh = ss.getSheetByName("precios");
  var unit = Number(data.precio) || 0;
  if (unit > 0) {
    preciosSh.getRange(2, 1, 1, 2).setValues([[1, unit]]);
    cfgSh.getRange(2, 1, cfgSh.getLastRow() - 1, 2).getValues().forEach(function (r, i) {
      if (r[0] === "precio_unitario") cfgSh.getRange(i + 2, 2).setValue(unit);
    });
  }

  var correo = String(data.compartir_con || "").trim();
  if (correo) {
    try {
      folder.addViewer(correo);
    } catch (e) {
      correo = "";
    }
  }

  var obj = {
    bingo_id: id, nombre: nombre, public_id: unico(8), hash_admin: unico(24), hash_cantador: unico(24),
    seed: randomHash_(16), spreadsheet_id: ss.getId(), folder_id: folder.getId(), created_at: ahora_(),
    activo: true, fecha_archivo: String(data.fecha_archivo || "").trim(), compartido_con: correo
  };
  writeObject_(sh, BINGOS_HEADERS_, obj);
  obj.vencido = false;
  return { bingo: bingoPublico_(obj) };
}

/**
 * Combos: el total de N cartones es la combinación de combos más barata que suma exactamente N
 * (ej. combos 1 = 3000 y 10 = 19000 → 21 cartones = 10 + 10 + 1 = 41000). Igual que precioCombos() en bingo.js.
 */
function precioCombos_(combos, unitario, cantidad) {
  var n = Math.floor(Number(cantidad) || 0);
  if (n <= 0) return 0;
  var lista = (combos || []).filter(function (c) { return c.cantidad > 0 && c.precio >= 0; });
  var unit = Number(unitario) || 0;
  if (!lista.some(function (c) { return c.cantidad === 1; }) && unit > 0) lista = lista.concat([{ cantidad: 1, precio: unit }]);
  if (!lista.length) return 0;
  var mejor = [0];
  for (var i = 1; i <= n; i++) {
    var m = Infinity;
    for (var j = 0; j < lista.length; j++) {
      var c = lista[j];
      if (c.cantidad <= i && mejor[i - c.cantidad] + c.precio < m) m = mejor[i - c.cantidad] + c.precio;
    }
    mejor.push(m);
  }
  if (!isFinite(mejor[n])) throw new Error("Con los combos actuales no se pueden vender " + n + " cartones. Agrega el combo de 1 cartón.");
  return mejor[n];
}

}

/** Busca cartones libres del fondo general (fuera de los rangos de sectores); amplía el lote si hace falta. */
function tomarLibres_(b, cantidad, sector) {
  var cfg = readConfig_(b);
  var habil = Number(cfg.cartones_habilitados) || LOTE_DEFAULT_;
  var usados = {};
  cartonesVendidos_(b).forEach(function (c) { usados[c.serie] = 1; });
  var rangos = sector ? [] : sectores_(b).filter(function (s) { return s.desde && s.hasta; });
  function enSector(n) {
    for (var i = 0; i < rangos.length; i++) if (n >= rangos[i].desde && n <= rangos[i].hasta) return true;
    return false;
  }
  var desde = sector ? sector.desde : 1;
  var hasta = sector ? sector.hasta : MAX_CARTONES_;
  var out = [];
  for (var n = desde; n <= hasta && out.length < cantidad; n++) {
    if (!sector && n > habil) {
      habil = Math.min(MAX_CARTONES_, habil + (Number(cfg.lote) || LOTE_DEFAULT_));
      if (n > habil) break;
    }
    if (!usados[n] && !enSector(n)) out.push(n);
  }
  if (out.length < cantidad) {
    throw new Error(sector
      ? "El sector «" + sector.nombre + "» no tiene " + cantidad + " cartones libres."
      : "No quedan " + cantidad + " cartones libres (máximo " + MAX_CARTONES_ + ").");
  }
  if (!sector && habil !== (Number(cfg.cartones_habilitados) || LOTE_DEFAULT_)) setConfigValue_(b, "cartones_habilitados", habil);
  return out;
}

function subcarpeta_(b, nombre) {

function decodeDataUrl_(dataUrl, nombre) {
  var m = String(dataUrl || "").match(/^data:([^;]+);base64,(.+)$/);
  if (!m) throw new Error("Archivo no válido.");
  var bytes = Utilities.base64Decode(m[2]);
  if (bytes.length > 8 * 1024 * 1024) throw new Error("El archivo pesa más de 8 MB.");
  return Utilities.newBlob(bytes, m[1], nombre);
}

function uploadArte_(b, data) {
  var ext = /png/.test(String(data.archivo || "")) ? "png" : "jpg";
  var blob = decodeDataUrl_(data.archivo, "arte-carton-" + Date.now() + "." + ext);
  if (!/^image\/(png|jpe?g|webp)$/.test(blob.getContentType())) throw new Error("El arte debe ser PNG, JPG o WEBP.");
  var file = subcarpeta_(b, "diseño").createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var cfg = readConfig_(b);
  if (cfg.arte_file_id) {
    try { DriveApp.getFileById(cfg.arte_file_id).setTrashed(true); } catch (e) { /* ya no existe */ }
  }
  setConfigValue_(b, "arte_file_id", file.getId());
  return { arte_file_id: file.getId(), arte_url: "https://lh3.googleusercontent.com/d/" + file.getId() };
}

function saveSectores_(b, lista) {
  var sh = sheetDe_(b, "sectores");
  var actuales = sectores_(b);
  var porId = {};
  actuales.forEach(function (s) { porId[s.sector_id] = s; });
  var nextId = nextId_(actuales, "sector_id");
  var limpios = lista.map(function (s) {
    var id = Number(s.sector_id) || nextId++;
    var prev = porId[id] || {};
    var desde = Number(s.desde) || 0;
    var hasta = Number(s.hasta) || 0;
    if (!String(s.nombre || "").trim()) throw new Error("Cada sector requiere un nombre.");
    if (desde < 0 || hasta < 0 || hasta > MAX_CARTONES_ || (desde && hasta && hasta < desde)) {
      throw new Error("Rango no válido en «" + s.nombre + "».");
    }
    var prevEnc = {};
    (prev.encargados || []).forEach(function (x) { prevEnc[x.id] = x; });
    var nextEnc = nextId_(prev.encargados || [], "id");
    var encargados = (Array.isArray(s.encargados) ? s.encargados : []).map(function (x) {
      var eid = Number(x.id) && prevEnc[Number(x.id)] ? Number(x.id) : nextEnc++;
      var nombreEnc = String(x.nombre || "").trim();
      if (!nombreEnc) throw new Error("Cada encargado de «" + s.nombre + "» requiere un nombre.");
      return {
        id: eid, nombre: nombreEnc, telefono: String(x.telefono || "").trim(),
        hash: (prevEnc[eid] && prevEnc[eid].hash) || b.public_id + "." + randomHash_(16),
        activo: x.activo === undefined ? true : bool_(x.activo)
      };
    });
    return {
      sector_id: id, nombre: String(s.nombre).trim(),
      encargado: encargados.map(function (x) { return x.nombre; }).join(", ") || String(s.encargado || "").trim(),
      telefono: String(s.telefono || "").trim(), desde: desde, hasta: hasta,
      hash: prev.hash || b.public_id + "." + randomHash_(16), activo: s.activo === undefined ? true : bool_(s.activo),
      encargados: JSON.stringify(encargados)
    };
  });
  for (var i = 0; i < limpios.length; i++) {
    for (var j = i + 1; j < limpios.length; j++) {
      var a = limpios[i], z = limpios[j];
      if (a.desde && z.desde && a.desde <= z.hasta && z.desde <= a.hasta) {
        throw new Error("Los rangos de «" + a.nombre + "» y «" + z.nombre + "» se cruzan.");
      }
    }
  }
  var maxHasta = 0;
  limpios.forEach(function (s) { maxHasta = Math.max(maxHasta, s.hasta); });
  var cfg = readConfig_(b);
  if (maxHasta > (Number(cfg.cartones_habilitados) || 0)) setConfigValue_(b, "cartones_habilitados", maxHasta);
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, SHEETS_.sectores.length).clearContent();
  if (limpios.length) {
    sh.getRange(2, 1, limpios.length, SHEETS_.sectores.length)
      .setValues(limpios.map(function (s) { return SHEETS_.sectores.map(function (h) { return s[h]; }); }));
  }
  return { sectores: sectores_(b) };
}

/**
 * El correo identifica al jugador: todas sus compras comparten el mismo link (token),
 * así juega todos sus cartones en una sola pantalla aunque los comprara en varias compras.
 */
function tokenPara_(b, correo) {
  var lista = entradas_(b);
  var clave = correoClave_(correo);
  if (clave) {
    var previa = lista.filter(function (e) { return e.token && correoClave_(e.correo) === clave; })[0];
    if (previa) return previa.token;
  }
  var usados = {};
  lista.forEach(function (e) { usados[e.token] = 1; });
  var t;
  do { t = b.public_id + "." + randomHash_(14); } while (usados[t]);
  return t;
}

function submitCompra_(b, data) {

  var telefono = String(data.telefono || "").replace(/[^\d+]/g, "");
  var correo = String(data.correo || "").trim();
  var cantidad = Math.floor(Number(data.cantidad) || 0);
  if (!nombre) throw new Error("Escribe tu nombre.");
  if (telefono.length < 8) throw new Error("Escribe un número de WhatsApp válido.");
  if (!correoValido_(correo)) throw new Error("Escribe un correo válido: con él se juntan todos tus cartones en un solo link.");
  if (cantidad < 1 || cantidad > MAX_POR_COMPRA_) throw new Error("La cantidad debe ser de 1 a " + MAX_POR_COMPRA_ + " cartones.");
  if (!data.comprobante) throw new Error("Adjunta el comprobante de pago.");
  var sectorId = Number(data.sector_id) || 0;
  var encargadoId = Number(data.encargado_id) || 0;
  var sectorElegido = sectores_(b).filter(function (s) { return s.sector_id === sectorId && s.activo; })[0];
  if (!sectorElegido) sectorId = 0;
  var encElegido = sectorElegido ? sectorElegido.encargados.filter(function (x) { return x.id === encargadoId; })[0] : null;
  if (!encElegido) encargadoId = 0;

  var blob = decodeDataUrl_(data.comprobante, "comprobante-" + Date.now() + "-" + nombre.replace(/[^\w]+/g, "_").slice(0, 30));
  var file = subcarpeta_(b, "comprobantes").createFile(blob);

  var lista = entradas_(b);
  var e = {
    entrada_id: nextId_(lista, "entrada_id"), fecha: ahora_(), nombre: nombre, telefono: telefono, correo: correo,
    modalidad: data.modalidad === "Presencial" ? "Presencial" : "Virtual", cantidad: cantidad,
    monto: precioPara_(b, cfg, cantidad), vendedor: encElegido ? encElegido.nombre : String(data.vendedor || "").trim(), sector_id: sectorId,
    metodo: String(data.metodo || "SINPE").trim(), comprobante_id: file.getId(), estado: "Pendiente",
    cartones: "", token: "", nota: "", email_enviado: false, wa_enviado: false, fecha_completada: "", encargado_id: encargadoId
  };
  guardarEntrada_(b, e);
  return { entrada_id: e.entrada_id, monto: e.monto, cantidad: cantidad };
}

  };

function approveEntrada_(b, data) {
  var e = entradaPorId_(b, data.entrada_id);
  if (e.estado !== "Pendiente") throw new Error("Solo se aprueban compras pendientes.");
  var sector = e.sector_id ? sectores_(b).filter(function (s) { return s.sector_id === e.sector_id; })[0] : null;
  var series;
  if (data.series) {
    series = parseSeries_(data.series);
    if (series.length !== e.cantidad) throw new Error("Indica exactamente " + e.cantidad + " cartones.");
    validarLibres_(b, series, null);
  } else {
    series = tomarLibres_(b, e.cantidad, sector && sector.desde && sector.hasta ? sector : null);
  }
  asignarCartones_(b, series, e, e.sector_id);
  e.cartones = series.join(",");
  e.token = e.token || tokenPara_(b, e.correo);
  e.estado = "Aprobada";
  e.nota = "";
  guardarEntrada_(b, e);
  delete e._row;
  return { entrada: e };
}

function validarLibres_(b, series, sector) {
  var cfg = readConfig_(b);
  var usados = {};
  cartonesVendidos_(b).forEach(function (c) { usados[c.serie] = 1; });
  series.forEach(function (n) {
    if (n < 1 || n > MAX_CARTONES_) throw new Error("La serie " + n + " no existe.");
    if (sector && (n < sector.desde || n > sector.hasta)) throw new Error("La serie " + n + " no es de tu sector (" + sector.desde + "–" + sector.hasta + ").");
    if (usados[n]) throw new Error("La serie " + n + " ya está vendida.");
  });
  var max = Math.max.apply(null, series);
  if (max > (Number(cfg.cartones_habilitados) || 0)) setConfigValue_(b, "cartones_habilitados", Math.min(MAX_CARTONES_, max));
}

function setEstadoEntrada_(b, id, estado, nota, liberar) {

  if (liberar && e.cartones) {
    liberarCartones_(b, e.entrada_id);

  e.estado = estado;
  if (nota !== undefined) e.nota = String(nota || "");
  e.fecha_completada = estado === "Completada" ? ahora_() : "";
  guardarEntrada_(b, e);
  delete e._row;
  return { entrada: e };
}

/**
 * Venta de un vendedor (organizador o encargado de sector): queda aprobada.
 * Con «series» anota los cartones impresos que entregó; con «cantidad» el sistema toma los siguientes libres
 * del bloque del sector (o del fondo general). El comprobante es opcional en efectivo.
 */
function venderPresencial_(b, data, sector, encargado) {
  var nombre = String(data.nombre || "").trim();
  var correo = String(data.correo || "").trim();
  if (!nombre) throw new Error("Escribe el nombre del comprador.");
  if (!correoValido_(correo)) throw new Error("Escribe el correo del comprador: con él se juntan todos sus cartones en un solo link.");
  var metodo = String(data.metodo || "Efectivo").trim();
  if (metodo !== "Efectivo" && !data.comprobante) throw new Error("Adjunta el comprobante del " + metodo + ".");
  var cfg = readConfig_(b);
  var desdeLink = Boolean(sector);
  if (!sector && Number(data.sector_id)) sector = sectores_(b).filter(function (s) { return s.sector_id === Number(data.sector_id); })[0] || null;
  var conBloque = sector && sector.desde && sector.hasta ? sector : null;
  var series;
  if (String(data.series || "").trim()) {
    series = parseSeries_(data.series);
    validarLibres_(b, series, desdeLink ? conBloque : null);
  } else {
    var cantidad = Math.floor(Number(data.cantidad) || 0);
    if (cantidad < 1 || cantidad > MAX_POR_COMPRA_) throw new Error("Indica las series entregadas o la cantidad de cartones (1 a " + MAX_POR_COMPRA_ + ").");
    series = tomarLibres_(b, cantidad, conBloque);
  }
  if (!series.length) throw new Error("Indica las series de los cartones vendidos.");
  var comprobanteId = "";
  if (data.comprobante) {
    var blob = decodeDataUrl_(data.comprobante, "comprobante-" + Date.now() + "-" + nombre.replace(/[^\w]+/g, "_").slice(0, 30));
    comprobanteId = subcarpeta_(b, "comprobantes").createFile(blob).getId();
  }
  var lista = entradas_(b);
  var e = {
    entrada_id: nextId_(lista, "entrada_id"), fecha: ahora_(), nombre: nombre,
    telefono: String(data.telefono || "").replace(/[^\d+]/g, ""), correo: correo,
    modalidad: data.modalidad === "Virtual" ? "Virtual" : "Presencial", cantidad: series.length,
    monto: data.monto !== undefined && data.monto !== "" ? Number(data.monto) || 0 : precioPara_(b, cfg, series.length),
    vendedor: String(encargado ? encargado.nombre : data.vendedor || (sector ? sector.encargado || sector.nombre : "")).trim(),
    sector_id: sector ? sector.sector_id : 0, metodo: metodo,
    comprobante_id: comprobanteId, estado: "Aprobada", cartones: series.join(","), token: "", nota: "",
    email_enviado: false, wa_enviado: false, fecha_completada: "", encargado_id: encargado ? encargado.id : Number(data.encargado_id) || 0
  };
  e.token = tokenPara_(b, correo);
  asignarCartones_(b, series, e, e.sector_id);
  guardarEntrada_(b, e);
  delete e._row;
  delete e.comprobante_id;
  e.tiene_comprobante = Boolean(comprobanteId);
  return { entrada: e };
}

function resumen_(entradas, vendidos, cfg) {
  var porEstado = { Pendiente: 0, Aprobada: 0, Completada: 0, Rechazada: 0 };
  var recaudado = 0, pendienteMonto = 0;
  var porSector = {};
  var montoSector = {};
  var porEncargado = {};
  var compradores = {};
  entradas.forEach(function (e) {
    porEstado[e.estado] = (porEstado[e.estado] || 0) + 1;
    if (e.estado === "Aprobada" || e.estado === "Completada") {
      recaudado += e.monto;
      var s = e.sector_id || 0;
      montoSector[s] = (montoSector[s] || 0) + e.monto;
      var k = s + ":" + (e.encargado_id || 0);
      var x = porEncargado[k] || (porEncargado[k] = { cartones: 0, monto: 0, compras: 0 });
      x.cartones += e.cantidad;
      x.monto += e.monto;
      x.compras += 1;
      compradores[correoClave_(e.correo) || "tel:" + e.telefono] = 1;
    }
    if (e.estado === "Pendiente") pendienteMonto += e.monto;
  });
  vendidos.forEach(function (c) {
    var k = c.sector_id || 0;
    porSector[k] = (porSector[k] || 0) + 1;
  });
  return {
    cartones_vendidos: vendidos.length, cartones_habilitados: Number(cfg.cartones_habilitados) || 0,
    compras: porEstado, recaudado: recaudado, por_aprobar: pendienteMonto, por_sector: porSector,
    monto_sector: montoSector, por_encargado: porEncargado, compradores: Object.keys(compradores).length
  };
}

/** Link de sector: el general ve todas las ventas del sector; el de un encargado, solo las suyas. */
function sectorResolve_(ctx) {
  var b = ctx.bingo, s = ctx.sector, enc = ctx.encargado;
  var cfg = readConfig_(b);
  var vendidos = cartonesVendidos_(b).filter(function (c) { return s.desde && c.serie >= s.desde && c.serie <= s.hasta; });
  var delSector = entradas_(b).filter(function (e) { return e.sector_id === s.sector_id; });
  var entradas = delSector.filter(function (e) { return !enc || e.encargado_id === enc.id; })
    .map(function (e) { delete e._row; delete e.comprobante_id; return e; }).reverse();
  return {
    sector: { sector_id: s.sector_id, nombre: s.nombre, encargado: enc ? enc.nombre : s.encargado, desde: s.desde, hasta: s.hasta },
    encargado: enc ? { id: enc.id, nombre: enc.nombre } : null,
    config: publicConfig_(b, cfg), plantilla_whatsapp: cfg.plantilla_whatsapp, sitio_url: cfg.sitio_url,
    sinpe_numero: cfg.sinpe_numero, sinpe_nombre: cfg.sinpe_nombre, iban: cfg.iban,
    precios: preciosObj_(b), precio_unitario: Number(cfg.precio_unitario) || 0,
    vendidos: vendidos.map(function (c) { return c.serie; }), entradas: entradas,
    sector_monto: delSector.filter(function (e) { return e.estado === "Aprobada" || e.estado === "Completada"; })
      .reduce(function (a, e) { return a + e.monto; }, 0)
  };
}

function getPlayer_(token) {
  var t = String(token || "").trim();
  var pub = t.split(".")[0];
  var b = bingoPorPublico_(pub);
  var todas = entradas_(b);
  var validas = function (x) { return x.estado === "Aprobada" || x.estado === "Completada"; };
  var base = todas.filter(function (x) { return x.token === t; });
  if (!base.length) throw new Error("Este link de cartones no es válido.");
  /* Mismo correo = mismo jugador: se juntan las compras aunque sean de antes de unificar los links. */
  var correos = {};
  base.forEach(function (x) { if (correoClave_(x.correo)) correos[correoClave_(x.correo)] = 1; });
  var suyas = todas.filter(function (x) { return validas(x) && (x.token === t || correos[correoClave_(x.correo)]); });
  if (!suyas.length) throw new Error("Esta compra todavía no está aprobada. Cuando se confirme el pago, aquí aparecen tus cartones.");
  var series = [];
  suyas.forEach(function (x) {
    x.cartones.split(",").map(Number).filter(Boolean).forEach(function (n) { if (series.indexOf(n) === -1) series.push(n); });
  });
  series.sort(function (a, z) { return a - z; });
  var cfg = readConfig_(b);
  return {
    config: publicConfig_(b, cfg), nombre: suyas[0].nombre.split(" ")[0],
    cartones: series, compras: suyas.length, vivo: estadoVivo_(b)
  };
}

function jsonDe_(raw, def) {
  try { return raw ? JSON.parse(String(raw)) : def; } catch (e) { return def; }
}

/**
 * Ronda: tipo «figuras» (varios premios, cada uno con su figura), «salado» (gana el cartón con más
 * casillas sin marcar) o «eliminacion» (cada bola elimina los cartones que la tienen; gana el último en pie).
 * Las rondas de la versión anterior (una figura y un premio) se leen como una ronda de figuras con un premio.
 */
function rondaObj_(r) {
  var tipo = TIPOS_RONDA_[String(r.tipo || "")] ? String(r.tipo) : "figuras";
  var premios = jsonDe_(r.premios, null);
  if (!Array.isArray(premios) || !premios.length) {
    premios = [{ id: 1, figura: tipo === "figuras" ? String(r.figura || "linea") : tipo, celdas: "", premio: String(r.premio || ""), ganadores: jsonDe_(r.ganadores, []) }];
  }
  return {
    ronda_id: Number(r.ronda_id), orden: Number(r.orden) || 0, nombre: String(r.nombre || ""), tipo: tipo,
    premios: premios.map(function (p, i) {
      return {
        id: Number(p.id) || i + 1, figura: String(p.figura || "linea"), celdas: String(p.celdas || ""),
        premio: String(p.premio || ""), ganadores: Array.isArray(p.ganadores) ? p.ganadores : []
      };
    }),
    estado: String(r.estado || "Pendiente"), bolas: String(r.bolas || "").split(",").map(Number).filter(Boolean),
    extra: jsonDe_(r.extra, {}) || {}, inicio: String(r.inicio || ""), fin: String(r.fin || ""), _row: r._row
  };
}

function rondas_(b) {
  return readObjects_(sheetDe_(b, "rondas")).map(rondaObj_).sort(function (a, z) { return a.orden - z.orden || a.ronda_id - z.ronda_id; });
}

function guardarRonda_(b, r) {
  var primero = r.premios[0] || {};
  var copia = {
    ronda_id: r.ronda_id, orden: r.orden, nombre: r.nombre, figura: primero.figura, premio: primero.premio,
    estado: r.estado, bolas: r.bolas.join(","), ganadores: JSON.stringify(primero.ganadores || []),
    inicio: r.inicio, fin: r.fin, tipo: r.tipo, premios: JSON.stringify(r.premios), extra: JSON.stringify(r.extra || {})
  };
  writeObject_(sheetDe_(b, "rondas"), SHEETS_.rondas, copia, r._row);
}

function limpiaRonda_(r) { var c = {}; Object.keys(r).forEach(function (k) { if (k !== "_row") c[k] = r[k]; }); return c; }

function gameGet_(b) {
  var cfg = readConfig_(b);
  return {
    rol: b._rol, seed: b.seed, config: publicConfig_(b, cfg), rondas: rondas_(b).map(limpiaRonda_),
    vivo: estadoVivo_(b), cartones_habilitados: Number(cfg.cartones_habilitados) || 0,
    vendidos: cartonesVendidos_(b).map(function (c) { return c.serie; })
  };
}

/** Texto «0,0;0,4;4,0» → casillas válidas de la figura personalizada. */
function celdasValidas_(txt) {
  var vistas = {};
  String(txt || "").split(";").forEach(function (p) {
    var m = p.match(/^([0-4]),([0-4])$/);
    if (m) vistas[m[1] + "," + m[2]] = 1;
  });
  return Object.keys(vistas).join(";");
}

function gameSaveRonda_(b, data) {
  var r = data.ronda || {};
  var tipo = TIPOS_RONDA_[r.tipo] ? r.tipo : "figuras";
  var lista = rondas_(b);
  var existente = r.ronda_id ? lista.filter(function (x) { return x.ronda_id === Number(r.ronda_id); })[0] : null;
  if (existente && existente.estado === "Terminada") throw new Error("Esta ronda ya terminó.");
  var previos = {};
  if (existente) existente.premios.forEach(function (p) { previos[p.id] = p; });
  var entrada = Array.isArray(r.premios) ? r.premios : [];
  var premios;
  if (tipo === "figuras") {
    if (!entrada.length) throw new Error("Agrega al menos un premio con su figura.");
    var siguiente = 1;
    entrada.forEach(function (p) { siguiente = Math.max(siguiente, (Number(p.id) || 0) + 1); });
    premios = entrada.map(function (p) {
      if (!FIGURAS_[p.figura] || TIPOS_RONDA_[p.figura] || (p.figura === "custom" && !celdasValidas_(p.celdas))) {
        throw new Error("Revisa la figura de cada premio.");
      }
      var id = Number(p.id) && previos[Number(p.id)] ? Number(p.id) : siguiente++;
      return {
        id: id, figura: p.figura, celdas: p.figura === "custom" ? celdasValidas_(p.celdas) : "",
        premio: String(p.premio || "").trim(), ganadores: previos[id] ? previos[id].ganadores : []
      };
    });
  } else {
    var unico = entrada[0] || {};
    premios = [{ id: 1, figura: tipo, celdas: "", premio: String(unico.premio || "").trim(), ganadores: previos[1] ? previos[1].ganadores : [] }];
  }
  var obj = existente || {
    ronda_id: nextId_(lista, "ronda_id"), orden: lista.length + 1, estado: "Pendiente", bolas: [], extra: {}, inicio: "", fin: ""
  };
  obj.nombre = String(r.nombre || "").trim() || ("Ronda " + obj.ronda_id);
  obj.tipo = tipo;
  obj.premios = premios;
  guardarRonda_(b, obj);
  if (obj.estado === "En juego") publicarVivo_(b, obj);
  return { rondas: rondas_(b).map(limpiaRonda_) };
}

function gameDeleteRonda_(b, data) {
  var r = rondas_(b).filter(function (x) { return x.ronda_id === Number(data.ronda_id); })[0];
  if (!r) throw new Error("Ronda no encontrada.");
  if (r.estado !== "Pendiente") throw new Error("Solo se borran rondas que no han empezado.");
  sheetDe_(b, "rondas").deleteRow(r._row);
  return { rondas: rondas_(b).map(limpiaRonda_) };
}

/** Cambia el orden de una ronda (dir = -1 sube, 1 baja). */
function gameMoverRonda_(b, data) {
  var lista = rondas_(b);
  var i = lista.map(function (x) { return x.ronda_id; }).indexOf(Number(data.ronda_id));
  var j = i + (Number(data.dir) < 0 ? -1 : 1);
  if (i < 0 || j < 0 || j >= lista.length) return { rondas: lista.map(limpiaRonda_) };
  var tmp = lista[i];
  lista[i] = lista[j];
  lista[j] = tmp;
  lista.forEach(function (x, k) {
    if (x.orden !== k + 1) { x.orden = k + 1; guardarRonda_(b, x); }
  });
  return { rondas: rondas_(b).map(limpiaRonda_) };
}

function rondaEnJuego_(b) {
  return rondas_(b).filter(function (x) { return x.estado === "En juego"; })[0] || null;
}

function gameAbrir_(b, data) {
  if (rondaEnJuego_(b)) throw new Error("Termina la ronda en juego antes de abrir otra.");
  var r = rondas_(b).filter(function (x) { return x.ronda_id === Number(data.ronda_id); })[0];
  if (!r) throw new Error("Ronda no encontrada.");
  r.estado = "En juego";
  r.inicio = r.inicio || ahora_();
  r.fin = "";
  anotarQuedan_(r, data);
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

/** En eliminación el cantador manda cuántos cartones vendidos siguen en pie (lo calcula su navegador). */
function anotarQuedan_(r, data) {
  if (r.tipo === "eliminacion" && data && data.quedan !== undefined && data.quedan !== null && data.quedan !== "") {
    r.extra = r.extra || {};
    r.extra.quedan = Math.max(0, Math.floor(Number(data.quedan) || 0));
  }
}

function gameCantar_(b, data) {
  var r = rondaEnJuego_(b);
  if (!r) throw new Error("Abre una ronda antes de cantar.");
  var bola = Math.floor(Number(data.bola));
  if (!(bola >= 1 && bola <= 75)) throw new Error("La bola debe ser de 1 a 75.");
  if (r.bolas.indexOf(bola) !== -1) throw new Error("La bola " + bola + " ya salió en esta ronda.");
  r.bolas.push(bola);
  anotarQuedan_(r, data);
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

function gameDeshacer_(b, data) {
  var r = rondaEnJuego_(b);
  if (!r || !r.bolas.length) throw new Error("No hay bolas para deshacer.");
  r.bolas.pop();
  anotarQuedan_(r, data);
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

function premioDe_(r, premioId) {
  var p = premioId ? r.premios.filter(function (x) { return x.id === Number(premioId); })[0] : r.premios.length === 1 ? r.premios[0] : null;
  if (!p) throw new Error("Elige a qué premio corresponde el ganador.");
  return p;
}

function gameGanador_(b, data) {
  var r = rondaEnJuego_(b);
  if (!r) throw new Error("No hay una ronda en juego.");
  var serie = Number(data.serie);
  if (!serie) throw new Error("Serie no válida.");
  var p = premioDe_(r, data.premio_id);
  if (p.ganadores.some(function (g) { return g.serie === serie; })) throw new Error("Ese cartón ya está registrado como ganador de este premio.");
  var venta = cartonesVendidos_(b).filter(function (c) { return c.serie === serie; })[0];
  p.ganadores.push({
    serie: serie, nombre: venta ? venta.comprador.split(" ")[0] : String(data.nombre || "").trim(),
    bolas: r.bolas.length, hora: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "HH:mm"),
    dato: data.dato !== undefined && data.dato !== null ? String(data.dato) : ""
  });
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

function gameQuitarGanador_(b, data) {
  var r = rondaEnJuego_(b);
  if (!r) throw new Error("No hay una ronda en juego.");
  var p = premioDe_(r, data.premio_id);
  p.ganadores = p.ganadores.filter(function (g) { return g.serie !== Number(data.serie); });
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

function gameCerrar_(b) {
  var r = rondaEnJuego_(b);
  if (!r) throw new Error("No hay una ronda en juego.");
  r.estado = "Terminada";
  r.fin = ahora_();
  guardarRonda_(b, r);
  return { ronda: limpiaRonda_(r), vivo: publicarVivo_(b, r) };
}

/** Para verificar: quién compró la serie (sin datos de contacto). */
function gameBuscar_(b, data) {
  var serie = Number(data.serie);
  var venta = cartonesVendidos_(b).filter(function (c) { return c.serie === serie; })[0];
  return { serie: serie, vendido: Boolean(venta), comprador: venta ? venta.comprador : "" };
}

/* ======================= Estado en vivo ======================= */

function premiosVivo_(r) {
  return r.premios.map(function (p) {
    var x = { i: p.id, f: p.figura, p: p.premio, g: p.ganadores.map(function (g) { return { s: g.serie, n: g.nombre, d: g.dato || undefined }; }) };
    if (p.celdas) x.c = p.celdas;
    return x;
  });
}

function vivoPayload_(b, r) {
  var terminadas = rondas_(b).filter(function (x) { return x.estado === "Terminada"; }).map(function (x) {
    return { n: x.nombre, tp: x.tipo, ps: premiosVivo_(x) };
  });
  if (!r) return { r: 0, e: "Espera", b: [], h: terminadas, t: Date.now() };
  var out = { r: r.ronda_id, n: r.nombre, tp: r.tipo, ps: premiosVivo_(r), e: r.estado, b: r.bolas, h: terminadas, t: Date.now() };
  if (r.tipo === "eliminacion" && r.extra && r.extra.quedan !== undefined) out.q = r.extra.quedan;
  return out;
}

function publicarVivo_(b, r) {
  var payload = vivoPayload_(b, r && r.estado === "En juego" ? r : null);
  if (r && r.estado === "Terminada") payload.ultima = { n: r.nombre, b: r.bolas.length };
  var json = JSON.stringify(payload);
  CacheService.getScriptCache().put("vivo_" + b.public_id, json, 21600);
  var vivoId = prop_("VIVO_SPREADSHEET_ID");
  if (vivoId) {
    var sh = SpreadsheetApp.openById(vivoId).getSheetByName("vivo");
    var ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues() : [];
    var fila = 0;
    for (var i = 0; i < ids.length; i++) if (String(ids[i][0]) === b.public_id) { fila = i + 2; break; }
    var valores = [[b.public_id, json, ahora_()]];
    if (fila) sh.getRange(fila, 1, 1, 3).setValues(valores);
    else sh.appendRow(valores[0]);
  }
  return payload;
}

function estadoVivo_(b) {
  var cached = CacheService.getScriptCache().get("vivo_" + b.public_id);
  if (cached) return JSON.parse(cached);
  var payload = vivoPayload_(b, rondaEnJuego_(b));
  CacheService.getScriptCache().put("vivo_" + b.public_id, JSON.stringify(payload), 21600);
  return payload;
}
